\section{Analytical inverse kinematics}
\label{sec:analytical-ik}

\subsection{Frames, geometry and the wrist target}
\label{sec:ik-position}
We seek the joint vectors $\vect{q}=(q_1,\ldots,q_6)^\trans$ giving a prescribed pose
of the endpoint frame $E$ (\texttt{link\_6}). Frame $0$ is \texttt{base\_link}; frames $1,\ldots,5$
are the physical URDF link frames, as in Section~\ref{sec:jacobian}.
The $q_i$ are URDF joint displacements from home. We use the DH lengths
from Table~\ref{tab:dh} and the fixed tilt $\gamma$ from Subsection~\ref{sec:poe}.

Let $W$ be the wrist centre, the origin of frame~5. The geometry is specified by
\begin{equation}
 \begin{aligned}
 {}^0\vect{t}_2\big|_{q_1=0}&=(a_1,0,d_1)^\trans,
 &{}^2\vect{t}_3&=(a_2,d_2,0)^\trans,\\
 {}^3\vect{p}_W&=(d_4,0,d_3)^\trans,
 &Q:={}^{1}R_2\big|_{q_2=0}&=R_x(\gamma).
 \end{aligned}
 \label{eq:ik-geometry}
\end{equation}
We retain $c_i=\cos q_i$, $s_i=\sin q_i$ and define
$c_\gamma=\cos\gamma$, $s_\gamma=\sin\gamma$.
The matrices $R_x,R_y,R_z$ are right-handed rotations.

All three wrist axes pass through $W$, so its position depends only on
$q_1,q_2,q_3$. The origin of $E$ lies a distance $d_6$ from $W$ along
its own $x$ axis. The subscript $d$ denotes a desired quantity, so the target
pose gives the desired wrist position
\begin{equation}
 \boxed{{}^0\vect{p}_{W,d}={}^0\vect{t}_{E,d}-d_6\,{}^0R_{E,d}\vect{e}_x
       =\begin{pmatrix}x_{w,d}&y_{w,d}&z_{w,d}\end{pmatrix}^{\trans}}.
 \label{eq:ik-wrist-target}
\end{equation}

\textbf{Wrist-centre position.}
Let $O$ be the point on axis~1 at shoulder height,
${}^0\vect{p}_O=d_1\vect{e}_z$. Combining the relative translations and rotations gives
\begin{equation}
 \begin{aligned}
 {}^0\vect{p}_W(\vect{q})={}&\underbrace{d_1\vect{e}_z}_{{}^0\vect{p}_O}+\underbrace{R_z(q_1)}_{{}^0R_1}
 \underbrace{\Biggl[
   \underbrace{a_1\vect{e}_x}_{{}^1\vect{t}_2-{}^1\vect{p}_O}
   +\underbrace{Q R_y(q_2)}_{{}^1R_2}
   \underbrace{\Biggl(
     \underbrace{\begin{pmatrix}a_2\\d_2\\0\end{pmatrix}}_{{}^2\vect{t}_3}
     +\underbrace{R_z(q_3)}_{{}^2R_3}
      \underbrace{\begin{pmatrix}d_4\\0\\d_3\end{pmatrix}}_{{}^3\vect{p}_W}
   \Biggr)}_{{}^2\vect{p}_W}
 \Biggr]}_{{}^1\vect{p}_W-{}^1\vect{p}_O}.
 \end{aligned}
 \label{eq:ik-position-map}
\end{equation}
This is the wrist-centre position obtained from the PoE model.
We evaluate it from the inside out. First, joint~3 gives the shoulder-to-wrist coordinates
in frame~2: ${}^2\vect{p}_W=(F_1,B,F_2)^\trans$, where
\begin{equation}
 F_1:=a_2+d_4c_3,\qquad B:=d_2+d_4s_3,\qquad F_2:=d_3.
 \label{eq:ik-auxiliary}
\end{equation}
Joint~2 leaves $B$ unchanged. Applying this rotation, then the fixed tilt
and shoulder offset, gives ${}^1\vect{p}_W-{}^1\vect{p}_O=(A,Y,H)^\trans$
\begin{equation}
 \begin{gathered}
 X=F_1c_2+F_2s_2,\qquad D=F_2c_2-F_1s_2,\\
 A=a_1+X,\qquad Y=c_\gamma B-s_\gamma D,\qquad H=s_\gamma B+c_\gamma D.
 \end{gathered}
 \label{eq:ik-intermediate-vectors}
\end{equation}
For an IK solution, the final base rotation gives
\begin{equation}
 x_{w,d}=Ac_1-Ys_1,\qquad y_{w,d}=As_1+Yc_1,\qquad z_{w,d}=d_1+H.
 \label{eq:ik-components}
\end{equation}

\subsection{Eliminating the first two joint angles}
We now form three \emph{known target quantities}:
\begin{equation}
 h:=z_{w,d}-d_1,\qquad \rho_{w,d}^2:=x_{w,d}^2+y_{w,d}^2,\qquad
 \Rcal:=\rho_{w,d}^2+h^2=\|{}^0\vect{p}_{W,d}-{}^0\vect{p}_O\|^2.
 \label{eq:ik-invariants}
\end{equation}
$h$ is the desired wrist height above $O$, $\rho_{w,d}$ is its distance to axis~1, and $\Rcal$ is
the wrist squared distance to $O$. None changes when joint~1 rotates. For a solution,
\eqref{eq:ik-components} gives $H=h$ and $\rho_{w,d}^2=A^2+Y^2$.

The $q_2$ rotation in \eqref{eq:ik-intermediate-vectors} preserves the norm:
\begin{equation}
 \underbrace{X^2+B^2+D^2}_{\|R_y(q_2)(F_1,B,F_2)^\trans\|^2}
 =\underbrace{F_1^2+B^2+F_2^2}_{\|(F_1,B,F_2)^\trans\|^2}
 \quad\Longrightarrow\quad X^2+D^2=F_1^2+F_2^2.
 \label{eq:ik-rotation-norm}
\end{equation}
The unchanged $y$ component $B$ cancels. Likewise,
$(Y,H)^\trans=\bigl(\begin{smallmatrix}c_\gamma&-s_\gamma\\s_\gamma&c_\gamma\end{smallmatrix}\bigr)(B,D)^\trans$
gives $Y^2+h^2=B^2+D^2$. Hence
\begin{equation}
 \begin{aligned}
 \Rcal&=(a_1+X)^2+B^2+D^2=2a_1X+F_3,\qquad
 h=s_\gamma B+c_\gamma D,\\
 F_3&:=a_1^2+F_1^2+F_2^2+B^2
     =L+2a_2d_4c_3+2d_2d_4s_3,\\
 L&:=a_1^2+a_2^2+d_2^2+d_3^2+d_4^2.
 \end{aligned}
 \label{eq:ik-invariant-equations}
\end{equation}
The expansion of $F_3$ uses $c_3^2+s_3^2=1$: $F_3$ depends only on $q_3$,
while $L$ is constant. With $\Rcal,h$ known and $a_1,c_\gamma$ nonzero,
a trial $q_3$ fixes
\begin{equation}
 X=\frac{\Rcal-F_3}{2a_1},\qquad
 D=\frac{h-s_\gamma B}{c_\gamma}.
 \label{eq:ik-XD}
\end{equation}
These required $X,D$ can come from a real rotation $q_2$ only if their norm
is correct. Substituting into \eqref{eq:ik-rotation-norm} and multiplying by
$4a_1^2$ gives us the \emph{closure condition}:
\begin{equation}
 \boxed{G(q_3):=(\Rcal-F_3)^2+
       \frac{4a_1^2}{c_\gamma^2}(h-s_\gamma B)^2
       -4a_1^2(F_1^2+d_3^2)=0.}
 \label{eq:ik-closure}
\end{equation}
Only $q_3$ remains unknown. For $\gamma=0$, $D=h$ and this becomes the
lecture's untilted closure, with terminal arm length $d_4$.

\subsection{Converting the closure into a quartic}
\label{sec:ik-quartic}
We use the half-angle substitution
\begin{equation}
 t=\tan(q_3/2),\qquad \Delta=1+t^2,\qquad
 c_3=\frac{1-t^2}{\Delta},\qquad s_3=\frac{2t}{\Delta}.
 \label{eq:ik-half-angle}
\end{equation}
Substituting into \eqref{eq:ik-closure} and multiplying by
$c_\gamma^2\Delta^2$ clears the denominators and gives
\begin{equation}
 \boxed{\begin{aligned}
 P(t)={}&c_\gamma^2\bigl[(\Rcal-L)\Delta-2a_2d_4(1-t^2)-4d_2d_4t\bigr]^2\\
       &+4a_1^2\bigl[h\Delta-s_\gamma(d_2\Delta+2d_4t)\bigr]^2
       -4a_1^2c_\gamma^2\bigl\{[a_2\Delta+d_4(1-t^2)]^2+d_3^2\Delta^2\bigr\}=0.
 \end{aligned}}
 \label{eq:ik-quartic}
\end{equation}
The expressions being squared have degree at most two, so $P$ has degree at most four.
Every real root gives a $q_3=2\arctan t$. Since $\Delta>0$ for real $t$,
clearing denominators adds no real roots. We must test $q_3=\pi$ separately in
\eqref{eq:ik-closure}, because its half-angle tangent is undefined. This missing point is included in the maximum of four distinct $q_3$ candidates.

\subsection{Recovering and checking the solutions}
\label{sec:ik-orientation}
\textbf{Arm angles.}
For each candidate $q_3$, we evaluate \eqref{eq:ik-auxiliary} for $F_1,F_2,B$ and
use \eqref{eq:ik-XD} for $X,D$. Combining their defining equations cancels
$s_2$ in $F_1X+F_2D$ and $c_2$ in $F_2X-F_1D$, giving us
\begin{equation}
 c_2=\frac{F_1X+F_2D}{F_1^2+F_2^2},\qquad
 s_2=\frac{F_2X-F_1D}{F_1^2+F_2^2},\qquad
 q_2=\operatorname{atan2}(s_2,c_2).
 \label{eq:ik-q2}
\end{equation}
The denominator is positive because $F_2=d_3\ne0$, and the closure ensures
$c_2^2+s_2^2=1$. Next we set $A=a_1+X$, $Y=c_\gamma B-s_\gamma D$ and recover
the base angle from the horizontal equations:
\begin{equation}
 q_1=\operatorname{atan2}(\underbrace{Ay_{w,d}-Yx_{w,d}}_{\rho_{w,d}^2s_1},\;
           \underbrace{Ax_{w,d}+Yy_{w,d}}_{\rho_{w,d}^2c_1}),\qquad \rho_{w,d}>0.
 \label{eq:ik-q1}
\end{equation}
The common positive factor $\rho_{w,d}^2$ does not change the angle.
If $\rho_{w,d}=0$, the wrist centre lies on axis~1 and $q_1$ is free, in which case we
choose a value in $[-\pi,\pi]$ and solve the wrist for that choice.

\textbf{Wrist angles.}
For each arm solution, we undo its orientation using the transpose, then solve
the remaining local $x$--$z$--$x$ wrist rotations:
\begin{equation}
 \begin{aligned}
 {}^0R_3&=R_z(q_1)Q R_y(q_2)R_z(q_3),\\
 {}^3R_{E,d}&=({}^0R_3)^\trans\,{}^0R_{E,d}
             =R_x(q_4)R_z(q_5)R_x(q_6)=:(r_{ij}).
 \end{aligned}
 \label{eq:ik-wrist-rotation}
\end{equation}
Here $r_{ij}$ is entry $(i,j)$ of ${}^3R_{E,d}$. Multiplying the wrist
rotations gives us five useful entries:
\begin{equation}
 r_{11}=c_5,\quad r_{21}=c_4s_5,\quad r_{31}=s_4s_5,\quad
 r_{12}=-s_5c_6,\quad r_{13}=s_5s_6.
 \label{eq:ik-wrist-entries}
\end{equation}
Let $\eta=\sqrt{r_{21}^2+r_{31}^2}=|s_5|$. For $\eta>0$, we choose each sign
$\varepsilon\in\{-1,+1\}$, so that $s_5=\varepsilon\eta$. Then
\begin{equation}
 \boxed{\begin{aligned}
 q_5&=\operatorname{atan2}(\varepsilon\eta,r_{11}),\\
 q_4&=\operatorname{atan2}(\varepsilon r_{31},\varepsilon r_{21}),\qquad
 q_6=\operatorname{atan2}(\varepsilon r_{13},-\varepsilon r_{12}).
 \end{aligned}}
 \label{eq:ik-wrist-solutions}
\end{equation}
We must use the same sign in all three formulas. It selects one of two
wrist configurations with the same orientation. If $\eta=0$, joints~4 and~6
are coaxial, so only a sum or difference is fixed:
\begin{equation}
 \begin{array}{lll}
 r_{11}=1: &q_5=0,&q_4+q_6=\operatorname{atan2}(r_{32},r_{22}),\\[2pt]
 r_{11}=-1:&q_5=\pi,&q_6-q_4=\operatorname{atan2}(r_{32},-r_{22}).
 \end{array}
 \label{eq:ik-wrist-singularity}
\end{equation}
We may set $q_4=0$ and recover $q_6$ from the appropriate relation.

\textbf{Assembly and validation.}
For every real $q_3$ candidate, we recover the arm and both wrist branches.
We represent angles in $[-\pi,\pi]$, enforce joint limits, remove duplicates
modulo $2\pi$, and check position and orientation by forward kinematics.
We use numerical tolerances for roots, singularities and pose errors.
Away from continuous singular families, at most four arm solutions each
have two wrist branches, so there are \textbf{at most eight full-pose solutions}.
